      chk_publicaciones_cilindrada
          CHECK (((cilindrada IS NULL) OR (char_length(TRIM(BOTH FROM cilindrada)) > 0)))
      chk_publicaciones_cilindros
          CHECK (((cilindros >= 0) AND (cilindros <= 16)))
      chk_publicaciones_color_otro
          CHECK (((char_length(TRIM(BOTH FROM color_otro)) >= 1) AND (char_length(TRIM(BOTH FROM
          color_otro)) <= 60)))
      chk_publicaciones_descripcion
          CHECK (((descripcion IS NULL) OR ((char_length(TRIM(BOTH FROM descripcion)) >= 1) AND
          (char_length(TRIM(BOTH FROM descripcion)) <= 2000))))
      chk_publicaciones_descripcion_problemas
          CHECK (((descripcion_problemas IS NULL) OR (char_length(descripcion_problemas) <=
          1000)))
      chk_publicaciones_deshabilitacion
          CHECK (((((estado_administrativo)::text = 'habilitada'::text) AND
          (motivo_deshabilitacion IS NULL) AND (fecha_deshabilitacion IS NULL)) OR
          (((estado_administrativo)::text = 'deshabilitada_administrador'::text) AND
          ((COALESCE(char_length(TRIM(BOTH FROM motivo_deshabilitacion)), 0) >= 1) AND
          (COALESCE(char_length(TRIM(BOTH FROM motivo_deshabilitacion)), 0) <= 1000)) AND
          (fecha_deshabilitacion IS NOT NULL))))
      chk_publicaciones_estado_administrativo
          CHECK (((estado_administrativo)::text = ANY ((ARRAY['habilitada'::character varying,
          'deshabilitada_administrador'::character varying])::text[])))
      chk_publicaciones_estado_publicacion
          CHECK (((estado_publicacion)::text = ANY ((ARRAY['activa'::character varying,
          'vendida'::character varying, 'deshabilitada'::character varying])::text[])))
      chk_publicaciones_fechas_aprobacion
          CHECK (((fecha_ultima_actualizacion_aprobada IS NULL) OR (fecha_primera_publicacion IS
          NOT NULL)))
      chk_publicaciones_horario
          CHECK (((horario_preferido_contacto IS NULL) OR (char_length(TRIM(BOTH FROM
          horario_preferido_contacto)) > 0)))
      chk_publicaciones_kilometraje
          CHECK (((kilometraje >= 0) AND (kilometraje <= 9999999)))
      chk_publicaciones_marca_otra
          CHECK (((char_length(TRIM(BOTH FROM marca_otra)) >= 1) AND (char_length(TRIM(BOTH FROM
          marca_otra)) <= 60)))
      chk_publicaciones_medio_venta
          CHECK (((medio_venta)::text = ANY ((ARRAY['dentro_app'::character varying,
          'fuera_app'::character varying])::text[])))
      chk_publicaciones_medio_venta_vendida
          CHECK ((((estado_publicacion)::text = 'vendida'::text) = (medio_venta IS NOT NULL)))
      chk_publicaciones_modelo_otro
          CHECK (((char_length(TRIM(BOTH FROM modelo_otro)) >= 1) AND (char_length(TRIM(BOTH
          FROM modelo_otro)) <= 60)))
      chk_publicaciones_modificaciones
          CHECK (((modificaciones IS NULL) OR ((char_length(TRIM(BOTH FROM modificaciones)) >=
          1) AND (char_length(TRIM(BOTH FROM modificaciones)) <= 1000))))
      chk_publicaciones_moneda
          CHECK ((moneda = ANY (ARRAY['MXN'::bpchar, 'USD'::bpchar])))
      chk_publicaciones_numero_pasajeros
          CHECK (((numero_pasajeros >= 1) AND (numero_pasajeros <= 99)))
      chk_publicaciones_numero_puertas
          CHECK (((numero_puertas >= 1) AND (numero_puertas <= 9)))
      chk_publicaciones_precio
          CHECK ((precio > (0)::numeric))
      chk_publicaciones_problemas
          CHECK ((((tiene_problemas IS NOT TRUE) AND (descripcion_problemas IS NULL)) OR
          ((tiene_problemas IS TRUE) AND (COALESCE(char_length(TRIM(BOTH FROM
          descripcion_problemas)), 0) > 0))))
      chk_publicaciones_propietarios_anteriores
          CHECK (((numero_propietarios_anteriores >= 0) AND (numero_propietarios_anteriores <=
          99)))
      chk_publicaciones_ruta_video
          CHECK (((ruta_video IS NULL) OR (char_length(TRIM(BOTH FROM ruta_video)) > 0)))
      chk_publicaciones_titulo
          CHECK ((char_length(TRIM(BOTH FROM titulo)) > 0))
      chk_publicaciones_transmision_otra
          CHECK (((char_length(TRIM(BOTH FROM transmision_otra)) >= 1) AND
          (char_length(TRIM(BOTH FROM transmision_otra)) <= 60)))

    Índices adicionales:

          USING (id_propietario = auth.uid())

TABLA public.solicitudes_destacado
----------------------------------------------------------------------------------------------------
    Solicitudes para destacar una publicación (1 mes) y su resolución.
    Filas: 6   RLS activado: sí

    Columnas:
      id_solicitud_destacado     bigint                       NOT NULL
                                 Identificador de la solicitud.
      id_publicacion             bigint                       NOT NULL
                                 Publicación que se solicita destacar.
      fecha_solicitud            timestamp with time zone     NOT NULL  DEFAULT now()
                                 Momento en que se creó la solicitud.
      estado_solicitud           character varying(12)        NOT NULL  DEFAULT 'pendiente'::character varying
                                 Estado actual de la solicitud.
      id_admin_resuelve          uuid                         NULL
                                 Administrador que aprueba o rechaza.
      motivo_rechazo             text                         NULL
                                 Motivo administrativo del rechazo.
      fecha_resolucion           timestamp with time zone     NULL
                                 Momento de aprobación o rechazo.
      fecha_inicio_destacado     timestamp with time zone     NULL
                                 Inicio del periodo aprobado.
      fecha_fin_destacado        timestamp with time zone     NULL
                                 Fin del periodo aprobado.
      fecha_cancelacion          timestamp with time zone     NULL
                                 Momento en que el propietario canceló una solicitud pendiente.
      aviso_vencimiento_enviado  boolean                      NOT NULL  DEFAULT false
                                 TRUE cuando ya se avisó al dueño que este destacado está por
                                 terminar.

          PRIMARY KEY (id_solicitud_destacado)

          FOREIGN KEY (id_publicacion) REFERENCES publicaciones(id_publicacion) ON DELETE
          RESTRICT

    CHECK:
      chk_solicitudes_destacado_campos
          CHECK (((((estado_solicitud)::text = 'pendiente'::text) AND (id_admin_resuelve IS
          NULL) AND (motivo_rechazo IS NULL) AND (fecha_resolucion IS NULL) AND
          (fecha_inicio_destacado IS NULL) AND (fecha_fin_destacado IS NULL) AND
          (fecha_cancelacion IS NULL)) OR (((estado_solicitud)::text = 'aprobada'::text) AND
          (id_admin_resuelve IS NOT NULL) AND (motivo_rechazo IS NULL) AND (fecha_resolucion IS
          NOT NULL) AND (fecha_inicio_destacado IS NOT NULL) AND (fecha_fin_destacado =
          (fecha_inicio_destacado + '1 mon'::interval)) AND (fecha_cancelacion IS NULL)) OR
          (((estado_solicitud)::text = 'rechazada'::text) AND (id_admin_resuelve IS NOT NULL)
          AND (COALESCE(char_length(TRIM(BOTH FROM motivo_rechazo)), 0) > 0) AND
          (fecha_resolucion IS NOT NULL) AND (fecha_inicio_destacado IS NULL) AND
          (fecha_fin_destacado IS NULL) AND (fecha_cancelacion IS NULL)) OR
          (((estado_solicitud)::text = 'cancelada'::text) AND (id_admin_resuelve IS NULL) AND
          (motivo_rechazo IS NULL) AND (fecha_resolucion IS NULL) AND (fecha_inicio_destacado IS
          NULL) AND (fecha_fin_destacado IS NULL) AND (fecha_cancelacion IS NOT NULL))))
      chk_solicitudes_destacado_estado
          CHECK (((estado_solicitud)::text = ANY ((ARRAY['pendiente'::character varying,
          'aprobada'::character varying, 'rechazada'::character varying, 'cancelada'::character
          varying])::text[])))
      chk_solicitudes_destacado_motivo_largo

----------------------------------------------------------------------------------------------------
FUNCIONES RPC QUE LLAMA LA APP
----------------------------------------------------------------------------------------------------

  contacto_vendedor(p_id_publicacion bigint)
      Devuelve: jsonb   Lenguaje: plpgsql   STABLE   SECURITY DEFINER
      Datos públicos del vendedor de una publicación. Teléfono, WhatsApp y correo solo con
      sesión iniciada.

  crear_publicacion(p_datos jsonb, p_fotos text[])
      Devuelve: bigint   Lenguaje: plpgsql   VOLATILE
      Crea una publicación nueva (vacía) y su primera propuesta con los datos y fotos; queda
      pendiente de revisión.

  editar_publicacion(p_id_publicacion bigint, p_datos jsonb, p_fotos text[])
      Devuelve: void   Lenguaje: plpgsql   VOLATILE   SECURITY DEFINER
      Edita una publicación propia: crea o reutiliza la propuesta pendiente/rechazada, reemplaza
      fotos (1 a 15, solo de la carpeta del dueño) y la manda a moderación.

  favoritos_mis_publicaciones()
      Devuelve: TABLE(id_publicacion bigint, total integer)   Lenguaje: sql   STABLE   SECURITY DEFINER
      Cuántos favoritos tiene cada publicación de la cuenta en sesión.

  marcar_publicacion_vendida(p_id_publicacion bigint, p_vendida boolean, p_medio_venta text DEFAULT NULL::text)
      Devuelve: void   Lenguaje: plpgsql   VOLATILE   SECURITY DEFINER
      El dueño marca su publicación como vendida (dentro_app/fuera_app) o la regresa a activa.
      Cancela solicitudes de destacado pendientes.

  mi_correo()
      Devuelve: text   Lenguaje: sql   STABLE   SECURITY DEFINER
      Correo de la cuenta en sesión (auth.users).

  quitar_dispositivo(p_token text)
      Devuelve: void   Lenguaje: sql   VOLATILE   SECURITY DEFINER
      Borra el token de Firebase al cerrar sesión.

  registrar_dispositivo(p_token text)
      Devuelve: void   Lenguaje: plpgsql   VOLATILE   SECURITY DEFINER
      Guarda el token de Firebase del teléfono para la cuenta en sesión (push).

  telefono_en_uso(p_numero text)
      Devuelve: boolean   Lenguaje: plpgsql   STABLE   SECURITY DEFINER
      TRUE si el número (10 dígitos, acepta +52) ya lo usa otra cuenta activa. Se usa en
      registro y Editar perfil.

  vendedor_estadisticas()
      Devuelve: jsonb   Lenguaje: plpgsql   STABLE   SECURITY DEFINER
      Indicadores del panel del vendedor (vistas, contactos, favoritos, publicaciones).

  Políticas de storage.objects:
    karsy_borrar_propio  (DELETE para authenticated)
        USING ((bucket_id = 'karsy'::text) AND ((storage.foldername(name))[2] =
        (auth.uid())::text))
    karsy_editar_propio  (UPDATE para authenticated)
        USING ((bucket_id = 'karsy'::text) AND ((storage.foldername(name))[2] =
        (auth.uid())::text))
    karsy_subir_propio  (INSERT para authenticated)
        WITH CHECK ((bucket_id = 'karsy'::text) AND ((storage.foldername(name))[1] = ANY
        (ARRAY['publicaciones'::text, 'perfiles'::text])) AND ((storage.foldername(name))[2] =
        (auth.uid())::text))
